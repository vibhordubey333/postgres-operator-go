Operator:
  kubectl get deployment {{ include "postgres-operator-go.fullname" . }} -n {{ .Release.Namespace }}
  kubectl logs -l app.kubernetes.io/name={{ include "postgres-operator-go.name" . }} -n {{ .Release.Namespace }} -f

CRD:
  kubectl get crd postgresdatabases.postgres.vibhordubey.com

Create a database:
  kubectl apply -f - <<EOF
  apiVersion: postgres.vibhordubey.com/v1alpha1
  kind: PostgresDatabase
  metadata:
    name: my-app-db
    namespace: {{ .Release.Namespace }}
  spec:
    databaseName: my_app
    storage:
      size: 10Gi
      storageClass: standard
  EOF

Watch provisioning:
  kubectl get postgresdatabase -n {{ .Release.Namespace }} -w

Get connection URL:
  kubectl get secret my-app-db-credentials -n {{ .Release.Namespace }} \
    -o jsonpath='{.data.DATABASE_URL}' | base64 -d
